using Unity.Pipeline.Commands;

/// <summary>Unity CLI command: `unity command #SCRIPTNAME# --target value`. Registers on the next recompile.</summary>
public static class #SCRIPTNAME#
{
    /// <summary>Set Required explicitly; it is not inferred. Append new parameters as optional ones at the end.</summary>
    [CliCommand("#SCRIPTNAME#", "Describe what this command does.")]
    public static object Run(
        [CliArg("target", "Describe what this argument selects.", Required = true)] string target,
        [CliArg("dry_run", "Report what would change without changing it.")] bool dryRun = false)
    {
        if (string.IsNullOrWhiteSpace(target))
        {
            throw new ArgumentException("Pass --target with a value.");
        }

        return new { target, dryRun };
    }
}
